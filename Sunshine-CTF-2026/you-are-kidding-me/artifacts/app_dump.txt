PROTECTED_FILES = frozenset(
    [os.path.realpath(os.path.join(KEYS_DIR, name))
     for name in ('reader.key', 'editor.key')]
    + [os.path.realpath(FLAG_PATH)]
)


def key_path(kid):
    return os.path.join(KEYS_DIR, kid)


def is_protected(kid):
    try:
        return os.path.realpath(key_path(kid)) in PROTECTED_FILES
    except (OSError, ValueError):
        return True


def load_key(kid):
    with open(key_path(kid), 'rb') as f:
        return f.read()


def current_reader():
    """Return {'sub', 'role'} for a valid pass cookie, or None.

    This is the ordinary read gate: it verifies the pass against its declared
    `kid` key (the same convention the whole site uses) and any valid pass —
    reader or editor — counts as signed in. A missing or tampered pass simply
    reads as "not signed in". The Editor's Desk does its own, louder
    verification (with the debug disclosure); this helper stays quiet.
    """
    token = request.cookies.get('token')
    if not token:
        return None
    try:
        header = jwt.get_unverified_header(token)
        key = load_key(header.get('kid', 'reader.key'))
        claims = jwt.decode(token, key, algorithms=['HS256'])
        return {'sub': claims.get('sub'), 'role': claims.get('role', 'reader')}
    except Exception:  # noqa: BLE001 - a bad/tampered/absent pass just reads as guest
        return None


@app.after_request
def no_store(resp):
    # Every rendered page reflects the (cookie-based) pass state, so a browser
    # must never serve a cached copy — otherwise a full article viewed while
    # holding a pass would reappear after sign-out, defeating the read gate.
    # Static assets (the public stylesheet) stay cacheable.
    if not request.path.startswith(('/static/', '/images/')):
        resp.headers['Cache-Control'] = 'no-store'
    return resp


@app.route('/images/<path:filename>')
def images(filename):
    """Serve the editorial plate photographs from the images folder."""
    return send_from_directory(IMAGES_DIR, filename)


@app.route('/')
def index():
    # Public front page of columns. Server-renders the reader's sign-in state
    # from the pass cookie so the header strip is accurate.
    return render_template('index.html', articles=ARTICLES, reader=current_reader())


@app.route('/article/<article_id>')
def article(article_id):
    """A single column. Reading the full text requires a valid reader pass."""
    art = next((a for a in ARTICLES if a['id'] == article_id), None)
    reader = current_reader()
    if art is None:
        return render_template('article.html', reader=reader, not_found=True), 404
    if not reader:
        # Subscribers-only: no pass, no full text.
        return render_template('article.html', article=art, reader=None, gated=True), 401
    return render_template('article.html', article=art, reader=reader)


@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        name = (request.form.get('name') or 'reader').strip() or 'reader'
        # Every visitor gets a low-privilege reader pass, signed with the
        # reader key and tagged with `kid: reader.key`.
        token = jwt.encode(
            {'sub': name, 'role': 'reader'},
            READER_KEY,
            algorithm='HS256',
            headers={'kid': 'reader.key'},
        )
        resp = make_response(redirect(url_for('index')))
        resp.set_cookie('token', token)
        return resp
    return render_template('login.html')


@app.route('/admin')
def admin():
    """The Editor's Desk. Only an `editor` pass may read the unpublished scoops."""
    token = request.cookies.get('token')
    if not token:
        return render_template('admin.html', need_pass=True), 401

    try:
        header = jwt.get_unverified_header(token)
    except Exception as e:  # noqa: BLE001 - malformed token
        return render_template('admin.html', error='BAD PASS :: ' + str(e)), 401

    kid = header.get('kid', 'reader.key')
    try:
        key = load_key(kid)
    except Exception as e:  # noqa: BLE001 - file not found etc.
        return render_template('admin.html', error='KEY LOAD FAILED :: ' + str(e), kid=kid), 401

    try:
        claims = jwt.decode(token, key, algorithms=['HS256'])
    except Exception as e:
        if is_protected(kid):
            return render_template(
                'admin.html',
                error='PASS REJECTED :: ' + str(e),
                kid=kid,
                withheld=True,
            ), 401
        disclosed = key.decode('utf-8', errors='replace')
        return render_template(
            'admin.html',
            error='PASS REJECTED :: ' + str(e),
            kid=kid,
            disclosed=disclosed,
        ), 401

    role = claims.get('role', 'reader')
    if role != 'editor':
        return render_template('admin.html', claims=claims, role=role, denied=True), 403

    return render_template('admin.html', claims=claims, role=role,
                           drafts=DRAFTS, flag=FLAG)


@app.route('/logout')
def logout():
    resp = make_response(redirect(url_for('index')))
    resp.delete_cookie('token')
    return resp
